class Clock:
    def __init__(self):
        self.value = 1000.
        self.sleeps = []
        self.interrupt_at = None
    def now(self):
        return self.value
    def sleep(self, seconds):
        self.sleeps.append(seconds)
        self.value += seconds
        if self.interrupt_at is not None and self.value >= self.interrupt_at:
            self.interrupt_at = None
            raise KeyboardInterrupt()


def safe_sample():
    return dict(timestamp=1000., power_query_ok=True, disk_free_bytes=10**12,
                power=dict(ac_connected=True, status='charging', percent=80),
                thermal=dict(available=True,state=0,name='nominal',temperature_c=None),
                process_query_ok=True, other_workers=[], cpu_percent=None, rss_bytes=None)


class Backend:
    def __init__(self, count=5):
        self.records = [dict(definition=dict(job_id=f'mock_{i}',kind='gru_trial',mapping='uniform_population'),status='NOT_STARTED') for i in range(count)]
        self.calls = []
        self.registrations = []
        self.interrupt = False
        self.after = None
    def rows(self):
        return self.records
    def verify_complete(self, job):
        assert next(r for r in self.records if r['definition']==job)['status']=='COMPLETE'
    def run(self, job, monitor):
        self.calls.append(job['job_id'])
        if self.interrupt:
            self.interrupt = False
            raise KeyboardInterrupt()
        row = next(r for r in self.records if r['definition']==job)
        row['status'] = 'COMPLETE'
        self.registrations.append(job['job_id'])
        if self.after:
            self.after(monitor)
        return dict(status='COMPLETE')


def controller(backend, clock, directory=None, probe=safe_sample):
    return SessionController(backend, probe, directory or OUT/'tests/sessions'/uuid.uuid4().hex, 10**9,
         now=clock.now, sleep=clock.sleep)


def test_two_jobs_per_session_order_and_no_final_sleep():
    backend, clock = Backend(), Clock()
    c = controller(backend,clock)
    result = c.run(max_sessions=3)
    assert backend.calls == [f'mock_{i}' for i in range(5)]
    assert [len(r['completed']) for r in result['sessions']] == [2,2,1]
    assert len(result['cooling_periods']) == 2
    assert sum(p['observed_wait_seconds'] for p in result['cooling_periods']) == 360
    assert max(clock.sleeps) <= 5


def test_bounded_default_one_session_and_restart_keeps_completed_jobs():
    backend, clock = Backend(4), Clock()
    c = controller(backend,clock)
    c.run()
    assert backend.calls == ['mock_0','mock_1']

    result = again.run()
    assert backend.calls == [f'mock_{i}' for i in range(4)]
    assert len(backend.registrations) == len(set(backend.registrations)) == 4
    assert result['cooling_periods'][0]['observed_wait_seconds'] == 180


def test_cooling_interrupt_restart_keeps_original_deadline():
    backend, clock = Backend(4), Clock()
    c = controller(backend,clock)
    c.run()
    deadline = c.state['cooling_until']
    clock.interrupt_at = clock.value+10
    with pytest.raises(KeyboardInterrupt):
        controller(backend,clock,c.directory).run()
    assert read(c.path)['cooling_until']==deadline
    restarted = controller(backend,clock,c.directory)
    restarted.run()
    assert sum(r['observed_wait_seconds'] for r in restarted.state['cooling_periods'])==180
    assert backend.calls == ['mock_0','mock_1','mock_2','mock_3']

def test_persistent_working_copy_reused_and_tampering_rejected(monkeypatch):
    from connectome.scripts.phase11 import session_launch as module
    directory=OUT/'tests/sessions'/uuid.uuid4().hex
    workspace=directory/'workspace'
    source='connectome/scripts/phase11/example.py'
    write(workspace/source,dict(unit_source=True))
    write(workspace/'earlier.json',dict(frozen=True))
    definitions={source:sha(workspace/source)}
    monkeypatch.setattr(module,'DIRECTORY',directory/'mode')
    monkeypatch.setattr(module,'sources',lambda:definitions)
    monkeypatch.setattr(module,'snapshot',lambda:(workspace,1))
    assert module.working_copy()==workspace
    monkeypatch.setattr(module,'snapshot',lambda:pytest.fail('Unnecessary execution copy reset'))
    assert module.working_copy()==workspace
    write(workspace/'earlier.json',dict(frozen=False))
    with pytest.raises(ValueError,match='bytes changed'):module.working_copy()

def test_controlled_batched_and_interrupted_training_matches_uninterrupted():
    from connectome.scripts.phase11.training import train_trial
    from connectome.scripts.phase10.gru import train_run
    from connectome.scripts.phase6.encoder import load_mapping
    rng=np.random.default_rng(812);dn=np.array(load_mapping()['channel_index'])<0;sets=[]
    for split in ('train','validation'):
        records=[]
        for k in range(20):
            x=rng.uniform(0,.2,(24,321))
            x[:,dn]=0.
            records.append(dict(scenario_id=f'fixture_{split}_{k}',group_id=f'fixture_{split}_{k}',split=split,cohort=split,
                                label=k%2,role='looming' if k%2 else 'static_initial',inputs=x))
        sets.append(records)
    root=OUT/'tests/sessions'/uuid.uuid4().hex
    class MathBackend(Backend):
        def __init__(self):
            super().__init__(2);self.results={};self.pause=True
            for row,rate in zip(self.records,(.0003,.001)):row['definition'].update(seed=55100,rate=rate)
        def run(self,job,monitor):
            net,report=train_trial(*sets,55100,job['rate'],root/job['job_id'],dict(mapping='synthetic_fixture'),
                                   smoke=True,smoke_epochs=3,pause_after_updates=1 if self.pause else None)
            self.pause=False
            if net is None:return dict(status='INTERRUPTED_CHECKPOINT_RESUMABLE')
            self.results[job['job_id']]=(net,report)
            return super().run(job,monitor)
    backend,clock=MathBackend(),Clock();c=controller(backend,clock,root/'journal')
    assert c.run()['status']=='INTERRUPTED_CHECKPOINT_RESUMABLE'
    controller(backend,clock,c.directory).run()
    for row in backend.records:
        job=row['definition'];expected,proof=train_run(*sets,55100,job['rate'],smoke=True,smoke_epochs=3,smoke_start=8)
        actual,report=backend.results[job['job_id']]
        assert report['history']==proof['history']
        for key in expected.p:np.testing.assert_array_equal(actual.p[key],expected.p[key])
    assert len(backend.registrations)==2
