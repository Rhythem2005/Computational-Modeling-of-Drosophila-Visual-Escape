"""One cooperating Phase 11 launcher at a time; crash-safe OS file lease."""
from contextlib import contextmanager
import fcntl
import json
import os
from .common import OUT


@contextmanager
def worker_lease(path=None):
    path = path or OUT/'worker.lock'
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('a+') as handle:
        try:
            fcntl.flock(handle, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError as error:
            raise RuntimeError('Another Phase 11 launcher holds the worker lease') from error
        try:
            handle.seek(0)
            handle.truncate()
            json.dump(dict(pid=os.getpid(), policy='one CPU-heavy Phase 11 launcher'), handle)
            handle.flush()
            yield
        finally:
            fcntl.flock(handle, fcntl.LOCK_UN)
